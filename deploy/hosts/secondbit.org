server {
	listen 80 default_server;
	listen [::]:80 default_server;

	include /etc/nginx/expires.conf;

	root /usr/share/nginx/html/secondbit.org;

	server_name secondbit.org;
	index index.html;

	location /health {
		expires -1;
		add_header Content-Type text/plain;
		return 200 'OK';
	}

	if ($http_x_forwarded_proto = "http") {
		return 301 https://secondbit.org$request_uri;
	}
}
